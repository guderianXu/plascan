cmake_minimum_required(VERSION 3.16)
project(PlaScanCore VERSION 1.1.0 LANGUAGES CXX)

message(STATUS "plascan_core: relying on vcpkg/conda/system CMAKE_PREFIX_PATH for LibTorch")

function(plascan_core_add_optional_module moduleDir moduleLabel)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${moduleDir}/CMakeLists.txt")
    add_subdirectory(${moduleDir})
    message(STATUS "plascan_core: building ${moduleLabel} module")
  else()
    message(WARNING "${moduleLabel} module not found")
  endif()
endfunction()

plascan_core_add_optional_module(feature_extractors "Feature Extractors")
plascan_core_add_optional_module(feature_match "FeatureMatch")
plascan_core_add_optional_module(camera "Camera")
plascan_core_add_optional_module(intersection "Intersection")
plascan_core_add_optional_module(bundle_adjust "BundleAdjust")
plascan_core_add_optional_module(overlap "Overlap")
plascan_core_add_optional_module(terrain "Terrain")
plascan_core_add_optional_module(sfm "SfM")
plascan_core_add_optional_module(mvs "MVS")
plascan_core_add_optional_module(dense_match "DenseMatch")
plascan_core_add_optional_module(mesh "Meshing")
